
# PostgreSQL — 쿠버네티스 배포 및 로컬 접속 매뉴얼
# (직접 작성한 YAML 매니페스트 + Argo CD GitOps 방식)
#
# 구성 파일 위치: postgres/ 디렉토리
#   - storageclass.yaml  : 로컬 스토리지 클래스 (postgres-local)
#   - pv.yaml            : kind-worker 노드 /mnt/postgres-data 를 10Gi PV로 등록
#   - pvc.yaml           : 위 PV를 요청하는 PVC
#   - configmap.yaml     : 초기 DB 자동 생성 SQL (airflow/mlflow/superset/grafana)
#   - statefulset.yaml   : postgres:16-alpine 파드 (role=postgres 노드에 고정)
#   - service.yaml       : ClusterIP 서비스 (postgres-postgresql:5432)
#
# 데이터 저장 경로 (kind extraMounts 로 연결)
#   kind-worker 노드 내부 : /mnt/postgres-data
#   Mac 로컬 머신        : /Users/danniel.kil/kind-data/postgres  ← 실제 데이터 보존 위치
#
# 사전 요구사항
#   - kind 클러스터 실행 중 (kind-worker 노드에 role=postgres 라벨 존재)
#   - Argo CD 설치 및 실행 중
#   - /Users/danniel.kil/kind-data/postgres 디렉토리 존재


# ──────────────────────────────────────────────
# 1. 사전 확인
# ──────────────────────────────────────────────

# 1-1) kind-worker 노드에 role=postgres 라벨 확인
kubectl get nodes --show-labels | grep role=postgres
# → kind-worker 노드에 라벨이 있어야 StatefulSet이 해당 노드에 스케줄됩니다.

# 1-2) 로컬 데이터 디렉토리 존재 확인
ls /Users/danniel.kil/kind-data/postgres
# → 없으면 생성: mkdir -p /Users/danniel.kil/kind-data/postgres

# 1-3) kind-worker 컨테이너 내부에 마운트 경로 존재 확인
docker exec kind-worker ls /mnt/postgres-data 2>/dev/null || echo "마운트 경로 없음 — 아래 명령으로 생성"
docker exec kind-worker mkdir -p /mnt/postgres-data


# ──────────────────────────────────────────────
# 2. Argo CD로 배포 (권장)
# ──────────────────────────────────────────────

# 2-1) Argo CD Application 등록 (최초 1회만)
kubectl apply -f argocd/application-postgres.yaml -n argocd

# 2-2) 배포 상태 확인
kubectl get application postgres-storage -n argocd
# SYNC STATUS: Synced / HEALTH STATUS: Healthy 가 되면 완료

# [주의] sync-wave 데드락 문제
# StorageClass 의 volumeBindingMode: WaitForFirstConsumer 설정 때문에
# PVC(wave 0)가 StatefulSet(wave 1)의 파드 없이는 Bound 상태가 되지 않습니다.
# Argo CD 는 PVC 가 Healthy(Bound) 상태여야 다음 wave 로 넘어가기 때문에
# 첫 배포 시 아래와 같이 wave 1 리소스를 직접 apply 해서 데드락을 해소해야 합니다.
#
# [해소 방법] wave 1 리소스 직접 적용
kubectl apply -f postgres/configmap.yaml -n postgres
kubectl apply -f postgres/service.yaml -n postgres
kubectl apply -f postgres/statefulset.yaml -n postgres
#
# → 이후 Argo CD 가 Synced/Healthy 로 전환되며 이후 변경사항은 자동 동기화됩니다.


# ──────────────────────────────────────────────
# 3. 배포 확인
# ──────────────────────────────────────────────

# 3-1) 파드 상태 확인 (Running 1/1 이 되어야 함)
kubectl get pods -n postgres

# 3-2) PV / PVC 바인딩 확인 (STATUS: Bound 가 되어야 함)
kubectl get pv,pvc -n postgres

# 3-3) 초기 DB 생성 확인 (airflow / mlflow / superset / grafana)
kubectl exec -n postgres postgres-postgresql-0 -- psql -U postgres -c "\l"


# ──────────────────────────────────────────────
# 4. 로컬 머신에서 접속
# ──────────────────────────────────────────────

# 4-1) 포트 포워딩 (터미널을 유지하는 동안 접속 가능)
#      로컬 5432 → 파드 5432
kubectl port-forward -n postgres svc/postgres-postgresql 5432:5432

# 4-2) 다른 터미널에서 접속 확인 (psql 설치된 경우)
psql -h localhost -U postgres -p 5432
# 비밀번호: postgres

# 4-3) DBeaver / DataGrip 등 GUI 툴 접속 정보
#   Host     : localhost
#   Port     : 5432
#   Database : postgres  (또는 airflow / mlflow / superset / grafana)
#   User     : postgres
#   Password : postgres

# [참고] psql 미설치 시 파드 내부에서 직접 접속
kubectl exec -it -n postgres postgres-postgresql-0 -- psql -U postgres


# ──────────────────────────────────────────────
# 5. 삭제
# ──────────────────────────────────────────────

# 5-1) Argo CD Application 삭제 (자동으로 연관 리소스도 삭제됨)
kubectl delete application postgres-storage -n argocd

# 5-2) 네임스페이스 및 PV 수동 정리 (Argo CD 삭제 후 남은 리소스)
kubectl delete namespace postgres
kubectl delete pv postgres-pv
kubectl delete storageclass postgres-local

# [주의] Mac 로컬 데이터는 삭제되지 않음 — 필요 시 수동 삭제
# rm -rf /Users/danniel.kil/kind-data/postgres/pgdata


# ──────────────────────────────────────────────
# 6. 로그 확인
# ──────────────────────────────────────────────

# 6-1) 파드가 아직 안 떴을 때 (Pending / ContainerCreating)
kubectl describe pod postgres-postgresql-0 -n postgres

# 6-2) 실행 중인 파드 로그 확인
kubectl logs postgres-postgresql-0 -n postgres -f

# 6-3) CrashLoopBackOff — 이전 컨테이너 로그
kubectl logs postgres-postgresql-0 -n postgres --previous
